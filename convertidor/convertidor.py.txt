import csv
import sys
from pathlib import Path

def txt_a_csv(entrada, salida="output.csv", separador=","):
    """Convierte un archivo .txt (una línea por registro) a CSV."""
    fuente = Path(entrada)
    if not fuente.exists():
        print(f"No existe: {entrada}")
        return

    lineas = fuente.read_text(encoding="utf-8").strip().split("\n")

    with open(salida, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f, delimiter=separador)
        for linea in lineas:
            writer.writerow([linea.strip()])

    print(f"Convertido: {entrada} → {salida} ({len(lineas)} líneas)")

if __name__ == "__main__":
    entrada = sys.argv[1] if len(sys.argv) > 1 else "datos.txt"
    txt_a_csv(entrada)
